# ==========================================
# 连连看一键打包上传脚本 (100% 本地化无敌版)
# ==========================================

Write-Host "1. 正在清理旧文件..." -ForegroundColor Cyan
Remove-Item -Recurse -Force build, docs -ErrorAction SilentlyContinue

Write-Host "2. 正在使用 Pygbag 打包..." -ForegroundColor Cyan
$env:PYTHONUTF8="1"
$job = Start-Job -ScriptBlock {
    $env:PYTHONUTF8="1"
    Set-Location $using:PWD
    python -m pygbag --build --ume_block 0 main.py
}
$job | Wait-Job -Timeout 15 | Out-Null
if ($job.State -eq 'Running') { Stop-Job $job }
Remove-Job $job -Force

Write-Host "3. 正在组装 docs 目录..." -ForegroundColor Cyan
New-Item -ItemType Directory -Path docs | Out-Null

if (Test-Path build\web\zhenhua_linkgame.tar.gz) {
    Copy-Item build\web\zhenhua_linkgame.tar.gz -Destination docs\ -Force
    Write-Host "✅ 核心包已提取" -ForegroundColor Green
} else {
    Write-Host "❌ 打包失败：找不到 .tar.gz 文件！" -ForegroundColor Red
    exit
}

# 强行去旧项目里拿 browserfs.min.js
$old_bfs_path = "F:\py_project\pythonProject\lianliankan\docs\browserfs.min.js"
if (Test-Path $old_bfs_path) {
    Copy-Item $old_bfs_path -Destination docs\browserfs.min.js -Force
    Write-Host "✅ 成功从旧项目复制 browserfs.min.js" -ForegroundColor Green
}

# 🌟 核心修复：从本地 Python 库提取 Pygbag 官方运行库，彻底断开外网 CDN！
Write-Host "4. 正在提取本地 Pygbag 运行库..." -ForegroundColor Cyan
$pygbag_cdn = python -c "import pygbag, os; print(os.path.join(os.path.dirname(pygbag.__file__), 'cdn'))"
if (Test-Path $pygbag_cdn) {
    # 把 pythons.js 和 cpython312 等所有底层文件复制到 docs
    Copy-Item -Path "$pygbag_cdn\*" -Destination docs -Recurse -Force
    Write-Host "✅ 本地运行库提取成功！完全脱离外网 CDN" -ForegroundColor Green
} else {
    Write-Host "⚠️ 未找到本地 Pygbag CDN 目录，网页可能依赖外网！" -ForegroundColor Yellow
}

# 写入终极自适应 index.html (指向本地 pythons.js)
$html_content = @"

<html lang="zh-CN">
<head>
    <link rel="icon" href="data:,">
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no">
    <title>振华连连看</title>
    <style>
        html, body {
            margin: 0; padding: 0; width: 100%; height: 100%;

            display: flex; justify-content: center; align-items: center;
        }
        canvas {
            max-width: 100vw;
            max-height: 100vh;
            image-rendering: -webkit-optimize-contrast;
            image-rendering: crisp-edges;
            touch-action: none;
        }
        #pyconsole, #terminal, #status, #progress, #infobox, #transfer, #system {
            display: none !important;
        }
    </style>
    <script src="browserfs.min.js"></script>
</head>
<body>
    <!-- 注意这里：把原来的外网链接改成了本地 pythons.js -->
    <script src="pythons.js" type="module" id="site" data-os="vtx,gui" async defer>
#<!--
import asyncio
import platform
from pathlib import Path
import tarfile

async def custom_site():
    platform.document.body.style.background = "#1a1e2a"
    platform.window.transfer.hidden = true
    platform.window.canvas.style.visibility = "visible"
    bundle = "zhenhua_linkgame"
    appdir = Path(f"/data/data/{bundle}")
    appdir.mkdir(exist_ok=True)
    import tarfile
    async with platform.fopen(f"{bundle}.tar.gz", "rb") as archive:
        tar = tarfile.open(fileobj=archive, mode="r:gz")
        tar.extractall(path=appdir.as_posix(), filter='tar')
        tar.close()
    platform.run_main(PyConfig, loaderhome= appdir / "assets", loadermain=None)
    import embed
    while embed.counter()<0:
        await asyncio.sleep(.1)
    main = appdir / "assets" / "main.py"
    if not platform.window.MM.UME:
        while not platform.window.MM.UME:
            await asyncio.sleep(.1)
    await TopLevel_async_handler.start_toplevel(platform.shell, console=False)
    __import__(__name__).__file__ = main
    await shell.source(main)
    platform.window.infobox.style.display = "none"
    platform.window.window_resize()
    shell.interactive()

asyncio.run(custom_site())
# --></script>
    <canvas id="canvas" width="1px" height="1px" oncontextmenu="event.preventDefault()" tabindex=1></canvas>
    <div id="infobox" style="display:none;">Loading...</div>
</body>
</html>
"@

Set-Content -Path docs\index.html -Value $html_content -Encoding UTF8
New-Item -ItemType File -Path docs\.nojekyll | Out-Null

Write-Host "5. 正在推送到 GitHub..." -ForegroundColor Cyan
git add .
git commit -m "Auto-deploy: 彻底本地化运行库，摆脱 CDN 依赖"
git push -f origin main
